/* Test case covering almost all C‑minus language features */

/* Global variable and array declarations */
int globalVar;
int globals[5];

/* Function declarations */
void init(void);
int factorial(int n);
int fib(int n, int results[]);
int sumArray(int arr[], int length);

/* Main entry point */
void main(void) {
    int i;
    int x;
    int y;
    int a[3];

    /* Initialize globals */
    init();

    /* Simple arithmetic and assignment */
    x = 3 + 4 * (2 - 1);
    y = x / 2;

    /* Call factorial */
    globalVar = factorial(5);       /* recursion */
    globals[0] = globalVar;

    /* Populate array a */
    i = 0;
    while (i < 3) {
        a[i] = i * i;
        i = i + 1;
    }

    /* Sum array a */
    globals[1] = sumArray(a, 3);

    /* Compute Fibonacci into globals[2..] */
    fib(6, globals);

    /* Conditional with break and continue */
    i = 0;
    while (i < 5) {
        if (globals[i] == 0) {
            i = i + 1;
            continue;
        } else if (globals[i] > 5) {
            break;
        }
        i = i + 1;
    }

    /* Return from main (void) */
    return;
}

/* Initialize globals and include comments */
void init(void) {
    /* Set globals to distinct values */
    globals[0] = 1;
    globals[1] = 2;
    globals[2] = 3;
    globals[3] = 4;
    globals[4] = 5;
    return;
}

/* Compute factorial recursively */
int factorial(int n) {
    if (n == 0) {
        return 1;
    } else {
        return n * factorial(n - 1);
    }
}

/* Compute Fibonacci sequence into results array */
int fib(int n, int results[]) {
    int idx;
    if (n <= 1) {
        results[2] = n;
        return n;
    }
    results[0] = fib(n - 1, results) + fib(n - 2, results);
    return results[0];
}

/* Sum elements of an integer array */
int sumArray(int arr[], int length) {
    int total;
    int j;
    total = 0;
    j = 0;
    while (j < length) {
        total = total + arr[j];
        j = j + 1;
    }
    return total;
}
